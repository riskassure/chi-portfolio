\begin{tikzpicture}[
  >=Stealth,
  shorten >=1pt,
  node distance=3.4cm,
  on grid,
  auto,
  every state/.style={minimum size=1.05cm}

  \path[->]
    (sigma) edge[bend left=20] node {$a$} (p)
    (p) edge[bend left=20] node {$b$} (sigma);
